Goal: A Rust CLI, totp-cli, that stores TOTP secrets in the OS credential store under a name and prints the current code for that name, so the agent can answer NetSuite 2FA prompts itself.

Architecture: Secrets live only in the OS credential store (macOS Keychain, Windows Credential Manager) behind a SecretStore trait with a file-backed implementation for tests and Linux. Non-secret parameters live in accounts.toml. RFC 6238 is implemented directly on hmac/sha1/sha2. The command layer mirrors netsuite-cli: clap derive, CliError with fixed exit codes, --json output, embedded SKILL.md, self-update from GitHub Releases.

use std::str::FromStr;

use hmac::{Hmac, Mac};
use serde::{Deserialize, Serialize};
use sha1::Sha1;
use sha2::{Sha256, Sha512};

use crate::error::CliError;

#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize, clap::ValueEnum)]
#[serde(rename_all = "lowercase")]
#[value(rename_all = "lowercase")]
pub enum Algorithm {
    Sha1,
    Sha256,
    Sha512,
}

pub fn totp(key: &[u8], unix_time: u64, period: u64, digits: u32, algorithm: Algorithm) -> String {
    hotp(key, unix_time / period, digits, algorithm)
}

pub fn expires_in(unix_time: u64, period: u64) -> u64 {
    period - (unix_time % period)
}

pub fn hotp(key: &[u8], counter: u64, digits: u32, algorithm: Algorithm) -> String {
    let digest = hmac_digest(key, &counter.to_be_bytes(), algorithm);
    // RFC 4226 §5.3 dynamic truncation: the low nibble of the last byte picks a 4-byte window.
    let offset = usize::from(digest[digest.len() - 1] & 0x0f);
    let window = [
        digest[offset] & 0x7f,
        digest[offset + 1],
        digest[offset + 2],
        digest[offset + 3],
    ];
    let truncated = u32::from_be_bytes(window);
    let code = truncated % 10u32.pow(digits);
    format!("{code:0width$}", width = digits as usize)
}

fn hmac_digest(key: &[u8], message: &[u8], algorithm: Algorithm) -> Vec<u8> {
    match algorithm {
        Algorithm::Sha1 => {
            let mut mac = Hmac::<Sha1>::new_from_slice(key).expect("HMAC accepts any key length");
            mac.update(message);
            mac.finalize().into_bytes().to_vec()
        }
        Algorithm::Sha256 => {
            let mut mac = Hmac::<Sha256>::new_from_slice(key).expect("HMAC accepts any key length");
            mac.update(message);
            mac.finalize().into_bytes().to_vec()
        }
        Algorithm::Sha512 => {
            let mut mac = Hmac::<Sha512>::new_from_slice(key).expect("HMAC accepts any key length");
            mac.update(message);
            mac.finalize().into_bytes().to_vec()
        }
    }
}

impl FromStr for Algorithm {
    type Err = CliError;

    fn from_str(text: &str) -> Result<Self, Self::Err> {
        match text.to_ascii_lowercase().replace('-', "").as_str() {
            "sha1" => Ok(Algorithm::Sha1),
            "sha256" => Ok(Algorithm::Sha256),
            "sha512" => Ok(Algorithm::Sha512),
            other => Err(CliError::Usage(format!(
                "unsupported algorithm '{other}'; use sha1, sha256 or sha512"
            ))),
        }
    }
}

use data_encoding::BASE32_NOPAD;
use zeroize::Zeroizing;

use crate::error::CliError;

/// A TOTP shared secret as normalised base32. Deliberately has no Debug/Display so it
/// cannot leak through logging or `{:?}` formatting.
pub struct Secret(Zeroizing<String>);

impl Secret {
    pub fn parse(input: &str) -> Result<Secret, CliError> {
        Self::decode(input).map_err(CliError::Usage)
    }

    pub fn from_stored(stored: &str) -> Result<Secret, CliError> {
        Self::decode(stored)
            .map_err(|message| CliError::Store(format!("stored secret is unreadable: {message}")))
    }

    pub fn as_base32(&self) -> &str {
        &self.0
    }

    pub fn key_bytes(&self) -> Zeroizing<Vec<u8>> {
        Zeroizing::new(
            BASE32_NOPAD
                .decode(self.0.as_bytes())
                .expect("Secret only holds base32 that decoded at construction"),
        )
    }

    fn decode(input: &str) -> Result<Secret, String> {
        let normalised: String = input
            .chars()
            .filter(|character| !character.is_whitespace() && *character != '-' && *character != '=')
            .map(|character| character.to_ascii_uppercase())
            .collect();
        if normalised.is_empty() {
            return Err("secret is empty".into());
        }
        BASE32_NOPAD
            .decode(normalised.as_bytes())
            .map_err(|_| "secret is not valid base32 (letters A-Z and digits 2-7)".to_string())?;
        Ok(Secret(Zeroizing::new(normalised)))
    }
}

pub fn parse(uri: &str) -> Result<ParsedUri, CliError> {
    let url = Url::parse(uri)
        .map_err(|parse_error| CliError::Usage(format!("invalid otpauth URI: {parse_error}")))?;
    if url.scheme() != "otpauth" {
        return Err(CliError::Usage("URI must start with otpauth://totp/".into()));
    }
    match url.host_str() {
        Some("totp") => {}
        Some("hotp") => return Err(CliError::Usage("HOTP (counter-based) URIs are not supported".into())),
        _ => return Err(CliError::Usage("URI must start with otpauth://totp/".into())),
    }

    let mut secret = None;
    let mut issuer = None;
    let mut digits = None;
    let mut period = None;
    let mut algorithm = None;
    for (key, value) in url.query_pairs() {
        match key.as_ref() {
            "secret" => secret = Some(Secret::parse(&value)?),
            "issuer" => issuer = Some(value.into_owned()),
            "digits" => digits = Some(parse_number::<u32>("digits", &value)?),
            "period" => period = Some(parse_number::<u64>("period", &value)?),
            "algorithm" => algorithm = Some(value.parse::<Algorithm>()?),
            _ => {}
        }
    }

    let secret = secret.ok_or_else(|| CliError::Usage("otpauth URI has no secret parameter".into()))?;
    Ok(ParsedUri {
        secret,
        issuer: issuer.or_else(|| issuer_from_label(url.path())),
        digits,
        period,
        algorithm,
    })
}

/// The label is `/Issuer:account` (percent-encoded); the issuer prefix is optional.
fn issuer_from_label(path: &str) -> Option<String> {
    let label = percent_decode_str(path.trim_start_matches('/')).decode_utf8_lossy();
    label
        .split_once(':')
        .map(|(issuer, _account)| issuer.trim().to_string())
        .filter(|issuer| !issuer.is_empty())
}

fn parse_number<Number: std::str::FromStr>(field: &str, value: &str) -> Result<Number, CliError> {
    value
        .parse::<Number>()
        .map_err(|_| CliError::Usage(format!("otpauth URI has a non-numeric {field} '{value}'")))
}

pub fn run(args: &AddArgs, config_dir: &Path, store: &dyn SecretStore) -> Result<Outcome, CliError> {
    validate_name(&args.name)?;
    let (secret, params) = resolve_secret_and_params(args)?;
    params.validate()?;

    let mut index = Index::load(config_dir)?;
    if index.accounts.contains_key(&args.name) && !args.force {
        return Err(CliError::Usage(format!(
            "account '{}' already exists; pass --force to replace it",
            args.name
        )));
    }

    store.set(&args.name, &secret)?;
    index.accounts.insert(args.name.clone(), params.clone());
    if let Err(save_error) = index.save(config_dir) {
        // Keep the store and the index in agreement: undo the secret we just wrote.
        let _ = store.delete(&args.name);
        return Err(save_error);
    }

    Ok(Outcome {
        human: format!("added {}", args.name),
        json: account_json(&args.name, &params),
    })
}

pub fn account_json(name: &str, params: &AccountParams) -> serde_json::Value {
    let mut value = json!({
        "name": name,
        "digits": params.digits,
        "period": params.period,
        "algorithm": params.algorithm,
    });
    if let Some(issuer) = &params.issuer {
        value["issuer"] = json!(issuer);
    }
    value
}

fn resolve_secret_and_params(args: &AddArgs) -> Result<(Secret, AccountParams), CliError> {
    let parsed_uri = args.uri.as_deref().map(otpauth::parse).transpose()?;
    let (secret, uri_issuer, uri_digits, uri_period, uri_algorithm) = match parsed_uri {
        Some(parsed) => (parsed.secret, parsed.issuer, parsed.digits, parsed.period, parsed.algorithm),
        None => (read_secret(args.secret.as_deref())?, None, None, None, None),
    };

        issuer: args.issuer.clone().or(uri_issuer),
        digits: args.digits.or(uri_digits).unwrap_or(DEFAULT_DIGITS),
        period: args.period.or(uri_period).unwrap_or(DEFAULT_PERIOD),
        algorithm: args.algorithm.or(uri_algorithm).unwrap_or(Algorithm::Sha1),
    };
    Ok((secret, params))
}

pub fn run(args: &CodeArgs, config_dir: &Path, store: &dyn SecretStore) -> Result<Outcome, CliError> {
    let index = Index::load(config_dir)?;
    let params = index.accounts.get(&args.name).ok_or_else(|| {
        CliError::Usage(format!("no account named '{}'; run `totp-cli list`", args.name))
    })?;
    if args.min_remaining >= params.period {
        return Err(CliError::Usage(format!(
            "--min-remaining must be less than the period ({}s)",
            params.period
        )));
    }
    let secret = store.get(&args.name)?.ok_or_else(|| {
        CliError::Store(format!(
            "secret for '{name}' is missing from the credential store; re-add it with `totp-cli add {name} --force`",
            name = args.name
        ))
    })?;

    let now = args.at.map(Ok).unwrap_or_else(current_unix_time)?;
    let time = effective_time(now, params.period, args.min_remaining);
    if time > now && args.at.is_none() {
        std::thread::sleep(Duration::from_secs(time - now));
    }

    let code = totp(&secret.key_bytes(), time, params.period, params.digits, params.algorithm);
    let remaining = expires_in(time, params.period);
    Ok(Outcome {
        human: code.clone(),
        json: json!({
            "name": args.name,
            "code": code,
            "expires_in": remaining,
            "period": params.period,
        }),
    })
}

/// Returns `now`, or the first second of the next period when the current code would
/// expire in fewer than `min_remaining` seconds.
pub fn effective_time(now: u64, period: u64, min_remaining: u64) -> u64 {
    let remaining = expires_in(now, period);
    if min_remaining > 0 && remaining < min_remaining {
        now + remaining
    } else {
        now
    }
}

fn current_unix_time() -> Result<u64, CliError> {
    SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|elapsed| elapsed.as_secs())
        .map_err(|clock_error| CliError::Io(format!("system clock is before 1970: {clock_error}")))
}

# totp-cli

Single-binary CLI that keeps TOTP secrets in the OS credential store (macOS Keychain,
Windows Credential Manager) and prints the current code for a named account. Built for
agents: the code is the only thing on stdout, errors go to stderr, exit codes are fixed.
Full reference: `totp-cli <subcommand> --help`.

## Pick the right command

| You need | Use |
|---|---|
| The current code for an enrolled account | `totp-cli code <name>` |
| A code that will still be valid while a form is filled and submitted | `totp-cli code <name> --min-remaining 5` |
| Code plus seconds until it rolls over | `totp-cli --json code <name>` → `{"name","code","expires_in","period"}` |
| Which accounts exist | `totp-cli list` (names only; secrets are never printed) |
| Enrol from a base32 secret | `totp-cli add <name> --secret <BASE32> [--issuer X]` |
| Enrol from a QR-code URI | `totp-cli add <name> --uri 'otpauth://totp/...'` |
| Replace a secret | `totp-cli add <name> ... --force` |
| Remove an account | `totp-cli remove <name>` |

## Recipe: a login page asks for a 2FA code

```bash
code=$(totp-cli code netsuite --min-remaining 5) || exit_code=$?
```

- Exit 0: `$code` is the 6-digit value; type it into the form and submit promptly.
- Exit 2 with "no account named": the account is not enrolled. **Ask the user** for the code
  (or to enrol it); never guess a name or try other accounts.
- Exit 3: the credential store refused or lost the secret. Report the stderr text to the user.

`--min-remaining 5` waits (at most a few seconds) for the next period if the current code
is about to expire, so the submitted code is never stale.

## Enrolling

Pass the secret with `--secret`, or run `totp-cli add <name>` with neither `--secret` nor
`--uri` to be prompted without echo. The secret is never echoed back or included in
`--json` output. Defaults are 6 digits, 30 seconds, SHA-1; `--uri` carries the service's
own values and explicit flags override them.

## Exit codes

| Code | Meaning | What to do |
|---|---|---|
| 0 | Success | Use stdout |
| 1 | Internal / I/O (e.g. unreadable accounts.toml) | Report stderr to the user |
| 2 | Usage: bad arguments, bad secret, unknown or duplicate account | Fix the call, or ask the user |
| 3 | Credential store unavailable, denied, or missing the secret | Report stderr; re-enrol with `add --force` if the user agrees |

## Storage

Secrets: OS credential store, service `totp-cli`, one item per account. Non-secret
parameters: `accounts.toml` in the per-user config dir (override with
`TOTP_CLI_CONFIG_DIR`). `TOTP_CLI_SECRET_STORE=file` swaps in an unencrypted
`secrets.toml` for tests and headless Linux only.

## Maintenance

`totp-cli update --check` reports a newer release; `totp-cli update` installs it and
refreshes this skill. `totp-cli skill install [--dir <path>]` rewrites the skill alone.

pub fn install(dir_override: Option<&Path>) -> Result<Value, CliError> {
    let config_base = config_base_dir();
    let base_exists = config_base.as_deref().is_some_and(Path::exists);
    let path = match resolve_target(dir_override, config_base.as_deref(), base_exists) {
        SkillTarget::Skip(reason) => {
            eprintln!("skill not installed ({reason}); place it yourself with `totp-cli skill install --dir <your skills dir>`");
            return Ok(skip_json(reason));
        }
        SkillTarget::Write(path) => path,
    };

    if is_symlink(&path) || path.parent().is_some_and(is_symlink) {
        return Ok(skip_json("symlink — skill tracks its source repo via git"));
    }
    if std::fs::read_to_string(&path).is_ok_and(|existing| existing == EMBEDDED_SKILL) {
        return Ok(installed_json(&path, false, Some("already current")));
    }
    if let Some(parent) = path.parent() {
        std::fs::create_dir_all(parent)
            .map_err(|io_error| CliError::Io(format!("cannot create {}: {io_error}", parent.display())))?;
    }
    std::fs::write(&path, EMBEDDED_SKILL)
        .map_err(|io_error| CliError::Io(format!("cannot write {}: {io_error}", path.display())))?;
    Ok(installed_json(&path, true, None))
}

pub fn resolve_target(dir_override: Option<&Path>, config_base: Option<&Path>, base_exists: bool) -> SkillTarget {

pub fn run(check_only: bool, skip_skill: bool) -> Result<Value, CliError> {

fn check_for_update() -> Result<Value, CliError> {
    let current_version = env!("CARGO_PKG_VERSION");
    let latest_release = build_updater()?
        .get_latest_release()
        .map_err(|update_error| CliError::Io(update_error.to_string()))?;
    Ok(json!({
        "current": current_version,
        "latest": latest_release.version,
        "updateAvailable": update_available(current_version, &latest_release.version)?,
    }))
}

/// Release tags carry a leading `v`; CARGO_PKG_VERSION never does. Compare as semver.
fn update_available(current: &str, latest: &str) -> Result<bool, CliError> {
    let latest_stripped = latest.strip_prefix(['v', 'V']).unwrap_or(latest);
    self_update::version::bump_is_greater(current, latest_stripped).map_err(|version_error| {
        CliError::Io(format!("cannot compare versions '{current}' and '{latest}': {version_error}"))
    })
}

fn install_update(skip_skill: bool) -> Result<Value, CliError> {
    // Resolve before the swap: on some platforms current_exe() is unreliable afterwards.
    let exe_path = std::env::current_exe();
    let status = build_updater()?
        .update()
        .map_err(|update_error| CliError::Io(update_error.to_string()))?;
    let mut result = json!({"updated": status.updated(), "version": status.version()});
    if !skip_skill && status.updated() {
        result["skill"] = match exe_path {
            Ok(exe) => refresh_skill_via_new_binary(&exe),
            Err(error) => skill_refresh_error(&format!("cannot locate updated binary: {error}")),
        };
    }
    Ok(result)
}

/// The running process still holds the OLD embedded skill, so run the freshly installed
/// binary's `skill install` instead. Never fatal.
fn refresh_skill_via_new_binary(exe: &Path) -> Value {
    match std::process::Command::new(exe).args(["skill", "install"]).output() {
        Ok(done) if done.status.success() => {
            if !done.stderr.is_empty() {
                eprint!("{}", String::from_utf8_lossy(&done.stderr));
            }
            serde_json::from_slice(&done.stdout)
                .map(drop_skill_name)
                .unwrap_or_else(|_| skill_refresh_error("skill install produced no JSON"))
        }
        Ok(done) => skill_refresh_error(&format!(
            "skill install exited {}: {}",
            done.status,
            String::from_utf8_lossy(&done.stderr).trim()
        )),
        Err(error) => skill_refresh_error(&format!("could not run skill install: {error}")),
    }
}

fn drop_skill_name(mut value: Value) -> Value {
    if let Some(map) = value.as_object_mut() {
        map.remove("skill");
    }
    value
}

fn skill_refresh_error(message: &str) -> Value {
    eprintln!("binary updated; skill refresh skipped — {message}");
    json!({"installed": false, "reason": message})
}

fn build_updater() -> Result<Box<dyn self_update::update::ReleaseUpdate>, CliError> {
    let mut updater = self_update::backends::github::Update::configure();
    updater
        .repo_owner(REPO_OWNER)
        .repo_name(REPO_NAME)
        .bin_name("totp-cli")
        .show_download_progress(false)
        .current_version(env!("CARGO_PKG_VERSION"));
    if let Ok(github_token) = std::env::var("GITHUB_TOKEN") {
        updater.auth_token(&github_token);
    }
    updater.build().map_err(|update_error| CliError::Io(update_error.to_string()))
}
